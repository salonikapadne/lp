public class EmployeeManager{
    public static void addEmployee(String empid, String name, String email, Date dob, String dept){
        Employee__c emp = new Employee__c();
        emp.Emp_ID__c = empid;
        emp.Name = name;
        emp.Email__c = email;
        emp.Birth_Date__c = dob;
        emp.Department__c = dept;
        
        insert emp;
        System.debug('Employee added succesfully');
    }
    
    public static void viewEmployee(){
        List<Employee__c> empList =[SELECT Emp_ID__c,Name,Email__c,Birth_Date__c,Department__c from Employee__c];
        
        for(Employee__c e:empList){
            System.debug(e.Emp_ID__c + '|' + e.Name + '|' + e.Email__c + '|' + e.Birth_Date__c + '|' + e.Department__c );
        }
    }
    
    public static void updateEmployee(String id,String newEmail){
        List<Employee__c> emplist = [SELECT id, Emp_ID__c,Email__c from Employee__c WHERE  Emp_ID__c =: id LIMIT 1];
        if(emplist.isEmpty()){System.debug('No entries Found');}
        Employee__c emp = emplist[0];
        emp.Email__c = newEmail;
        update emp;
        System.debug('Updated Successfully');
    }
    
    public static void deleteEmployee(String id){
        List<Employee__c> emplist = [SELECT id, Emp_ID__c,Email__c from Employee__c WHERE  Emp_ID__c =: id LIMIT 1];
        if(emplist.isEmpty()){System.debug('No entries Found');}
        delete emplist[0];
        System.debug('Deleted');
    }
}
