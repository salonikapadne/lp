empmanager.js
import { LightningElement } from 'lwc';
import { ShowToastEvent } from 'lightning/platformShowToastEvent';

export default class Empmanager extends LightningElement {

    empName = '';
    empId = '';
    salary = '';
    email = '';
    department = '';
    joiningDate = '';

    employeeIds = [];

    departmentOptions = [
        { label: 'HR', value: 'HR' },
        { label: 'IT', value: 'IT' },
        { label: 'Finance', value: 'Finance' },
        { label: 'Marketing', value: 'Marketing' }
    ];

    handleName(event) {
        this.empName = event.target.value;
    }

    handleId(event) {
        this.empId = event.target.value;
    }

    handleSalary(event) {
        this.salary = event.target.value;
    }

    handleEmail(event) {
        this.email = event.target.value;
    }

    handleDepartment(event) {
        this.department = event.target.value;
    }

    handleJoiningDate(event) {
        this.joiningDate = event.target.value;
    }

    saveEmployee() {

        if (!this.empName || this.empName.trim().length < 3) {
            this.showToast(
                'Error',
                'Employee Name must contain at least 3 characters',
                'error'
            );
            return;
        }

        if (this.empId <= 0) {
            this.showToast(
                'Error',
                'Employee ID must be greater than 0',
                'error'
            );
            return;
        }

        if (this.employeeIds.includes(this.empId)) {
            this.showToast(
                'Error',
                'Employee ID must be unique',
                'error'
            );
            return;
        }

        if (this.salary <= 10000 || this.salary >= 500000) {
            this.showToast(
                'Error',
                'Salary must be between 10,000 and 500,000',
                'error'
            );
            return;
        }

        const emailPattern = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;

        if (!emailPattern.test(this.email)) {
            this.showToast(
                'Error',
                'Invalid Email Format',
                'error'
            );
            return;
        }

        if (!this.department) {
            this.showToast(
                'Error',
                'Please select a department',
                'error'
            );
            return;
        }

        let today = new Date();
        let joinDate = new Date(this.joiningDate);

        if (joinDate > today) {
            this.showToast(
                'Error',
                'Joining Date cannot be future date',
                'error'
            );
            return;
        }

        this.employeeIds.push(this.empId);

        this.showToast(
            'Success',
            'Employee Saved Successfully',
            'success'
        );
    }

    showToast(title, message, variant) {

        this.dispatchEvent(
            new ShowToastEvent({
                title,
                message,
                variant
            })
        );
    }
}
