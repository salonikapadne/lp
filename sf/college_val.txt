import { LightningElement } from 'lwc';
import { ShowToastEvent } from 'lightning/platformShowToastEvent';

export default class Collegemanager extends LightningElement {

    studentName = '';
    rollNumber = '';
    marks = '';
    studentEmail = '';

    facultyName = '';
    facultyId = '';
    salary = '';
    department = '';
    joiningDate = '';

    students = [];
    faculties = [];

    departmentOptions = [
        { label: 'Computer', value: 'Computer' },
        { label: 'IT', value: 'IT' },
        { label: 'Mechanical', value: 'Mechanical' },
        { label: 'Civil', value: 'Civil' }
    ];

    handleStudentName(event) {
        this.studentName = event.detail.value || '';
    }

    handleRollNumber(event) {
        this.rollNumber = Number(event.detail.value || 0);
    }

    handleMarks(event) {
        this.marks = Number(event.detail.value || 0);
    }

    handleStudentEmail(event) {
        this.studentEmail = event.detail.value || '';
    }

    handleFacultyName(event) {
        this.facultyName = event.detail.value || '';
    }

    handleFacultyId(event) {
        this.facultyId = Number(event.detail.value || 0);
    }

    handleSalary(event) {
        this.salary = Number(event.detail.value || 0);
    }

    handleDepartment(event) {
        this.department = event.detail.value || '';
    }

    handleJoiningDate(event) {
        this.joiningDate = event.detail.value || '';
    }

    saveStudent() {

        if (!this.studentName) {
            this.showToast('Error', 'Student Name cannot be blank', 'error');
            return;
        }

        if (!this.rollNumber || Number.isNaN(this.rollNumber) || this.rollNumber <= 0) {
            this.showToast('Error', 'Roll Number must be greater than 0', 'error');
            return;
        }

        if (Number.isNaN(this.marks) || this.marks < 0 || this.marks > 100) {
            this.showToast('Error', 'Marks should be between 0 and 100', 'error');
            return;
        }

        if (!this.studentEmail || !this.studentEmail.includes('@')) {
            this.showToast('Error', 'Email must contain @ symbol', 'error');
            return;
        }

        this.showToast('Success', 'Student Record Saved', 'success');
    }

    saveFaculty() {

        if (!this.facultyName || this.facultyName.trim().length < 3) {
            this.showToast('Error', 'Faculty Name must contain at least 3 characters', 'error');
            return;
        }

        if (!this.facultyId || Number.isNaN(this.facultyId) || this.facultyId <= 0) {
            this.showToast('Error', 'Faculty ID must be greater than 0', 'error');
            return;
        }

        let existingFaculty = this.faculties.find(
            fac => fac.facultyId == this.facultyId
        );

        if (existingFaculty) {
            this.showToast('Error', 'Faculty ID must be unique', 'error');
            return;
        }

        if (Number.isNaN(this.salary) || this.salary <= 10000 || this.salary >= 500000) {
            this.showToast('Error', 'Salary must be between 10,000 and 500,000', 'error');
            return;
        }

        if (!this.department) {
            this.showToast('Error', 'Select Department', 'error');
            return;
        }

        let today = new Date();
        let joinDate = new Date(this.joiningDate);

        if (joinDate > today) {
            this.showToast('Error', 'Joining Date cannot be future date', 'error');
            return;
        }

        this.showToast('Success', 'Faculty Record Saved', 'success');
    }

    showToast(title, message, variant) {

        this.dispatchEvent(
            new ShowToastEvent({
                title,
                message,
                variant
            })
        );
    }
}
