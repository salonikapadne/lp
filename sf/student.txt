
public class StudentController {

    public Student__c student { get; set; }
    public List<Student__c> students { get; set; }

    public String editRollNo { get; set; }

    public StudentController() {
        student = new Student__c();
        loadStudents();
    }

    // LOAD
    public void loadStudents() {
        students = [
            SELECT Id, Name, Roll_No__c, Class__c, Phone__c
            FROM Student__c
            ORDER BY CreatedDate DESC
        ];
    }

    // CREATE + UPDATE
    public void saveStudent() {

        if(editRollNo == null) {
            StudentManager.addStudent(
                student.Roll_No__c,
                student.Name,
                student.Class__c,
                student.Phone__c
            );
        } else {
            StudentManager.updateStudent(
            student.Roll_No__c,
            student.Phone__c
        );
        }

        student = new Student__c();
        editRollNo = null;
        loadStudents();
    }

    // DELETE
    public void deleteStudent() {
        String roll = ApexPages.currentPage().getParameters().get('roll');
        StudentManager.deleteStudent(roll);
        loadStudents();
    }

    // EDIT (prefill form)
    public void editStudent() {
        String roll = ApexPages.currentPage().getParameters().get('roll');

        Student__c s = [
            SELECT Name, Roll_No__c, Class__c, Phone__c
            FROM Student__c
            WHERE Roll_No__c = :roll
            LIMIT 1
        ];

        student = s;
        editRollNo = roll;
    }

    // READ (if you want controller-managed list)
    public List<Student__c> getStudentList() {
        return students;
    }
}

public class StudentManager{
    public static void addStudent(String rollno,String name,String cls,String phn){
        Student__c s = new Student__c();
        
        s.Roll_No__c = rollno;
        s.Name = name;
        s.Class__c = cls;
        s.Phone__c = phn;
        
        insert s;
        System.debug('Added successfully');
    }
    
    public static void viewStudent(){
       List<Student__c> studlist =[SELECT Roll_No__c,Name,Class__c,Phone__c from Student__c];
       if(studlist.isEmpty()){System.debug('List is empty');}
       for(Student__c s:studlist){
           System.debug(s.Roll_No__c + '|' + s.Name + '|' + s.Class__c + '|' + s.Phone__c);
       }
    }
    
    public static void updateStudent(String rollNo, String newPhone) {

        List<Student__c> s = [
            SELECT Id, Phone__c
            FROM Student__c
            WHERE Roll_No__c = :rollNo
            LIMIT 1
        ];
    
        if(s.isEmpty()){
            System.debug('No records');
            return;
        }
    
        Student__c st = s[0];
    
        st.Phone__c = newPhone;
    
        update st;
    
        System.debug('Updated');
    }
    public static void deleteStudent(String roll){
        List<Student__c> studlist = [SELECT id,Roll_No__c,Name FROM Student__c where Roll_No__c =: roll LIMIT 1];
        if(studlist.isEmpty()){System.debug('List is empty');}
        
        delete studlist[0];
        System.debug('Deleted');
    }
}
